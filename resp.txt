import json
import requests
import pandas as pd
import xml.etree.ElementTree as ET


def _xml_to_dict(elem):
    """Recursively flatten an XML element into dict/list (repeated tags become lists)."""
    children = list(elem)
    if not children:
        text = elem.text.strip() if elem.text else None
        return {"@attrib": elem.attrib, "text": text} if elem.attrib else text

    result = {}
    for child in children:
        data = _xml_to_dict(child)
        if child.tag in result:
            if not isinstance(result[child.tag], list):
                result[child.tag] = [result[child.tag]]
            result[child.tag].append(data)
        else:
            result[child.tag] = data
    if elem.attrib:
        result["@attrib"] = elem.attrib
    return result


def _json_or_none(elem, tag):
    node = elem.find(tag)
    return json.dumps(_xml_to_dict(node)) if node is not None else None


def fetch_qualys_kb(username, password, platform_url="https://qualysapi.qualys.com", truncation_limit=1000):
    """Fetch all Qualys KnowledgeBase (QID) records via API 4.0, return as DataFrame."""
    url = f"{platform_url}/api/4.0/fo/knowledge_base/vuln/"
    headers = {"X-Requested-With": "python-requests"}
    params = {"action": "list", "details": "All", "truncation_limit": truncation_limit}
    auth = (username, password)

    records = []
    id_min = None

    while True:
        if id_min:
            params["id_min"] = id_min

        resp = requests.get(url, headers=headers, params=params, auth=auth)
        resp.raise_for_status()
        root = ET.fromstring(resp.content)

        vulns = root.findall(".//VULN")
        if not vulns:
            break

        for v in vulns:
            records.append({
                "QID": v.findtext("QID"),
                "VULN_TYPE": v.findtext("VULN_TYPE"),
                "SEVERITY_LEVEL": v.findtext("SEVERITY_LEVEL"),
                "TITLE": v.findtext("TITLE"),
                "CATEGORY": v.findtext("CATEGORY"),
                "PATCHABLE": v.findtext("PATCHABLE"),
                "PUBLISHED_DATETIME": v.findtext("PUBLISHED_DATETIME"),
                "LAST_SERVICE_MODIFICATION_DATETIME": v.findtext("LAST_SERVICE_MODIFICATION_DATETIME"),
                "CODE_MODIFIED_DATETIME": v.findtext("CODE_MODIFIED_DATETIME"),
                "PATCH_PUBLISHED_DATE": v.findtext("PATCH_PUBLISHED_DATE"),
                "TECHNOLOGY": v.findtext("TECHNOLOGY"),
                "DIAGNOSIS": v.findtext("DIAGNOSIS"),
                "CONSEQUENCE": v.findtext("CONSEQUENCE"),
                "SOLUTION": v.findtext("SOLUTION"),
                "CVSS_BASE": v.findtext("CVSS/BASE"),
                "CVSS3_BASE": v.findtext("CVSS_V3/BASE"),
                "SOFTWARE_LIST": _json_or_none(v, "SOFTWARE_LIST"),
                "VENDOR_REFERENCE_LIST": _json_or_none(v, "VENDOR_REFERENCE_LIST"),
                "CVE_LIST": _json_or_none(v, "CVE_LIST"),
                "BUGTRAQ_LIST": _json_or_none(v, "BUGTRAQ_LIST"),
                "COMPLIANCE_LIST": _json_or_none(v, "COMPLIANCE_LIST"),
                "THREAT_INTELLIGENCE": _json_or_none(v, "THREAT_INTELLIGENCE"),
                "CORRELATION": _json_or_none(v, "CORRELATION"),
            })

        warning = root.find(".//WARNING/URL")
        if warning is not None and "id_min=" in warning.text:
            id_min = warning.text.split("id_min=")[1].split("&")[0]
        else:
            break

    return pd.DataFrame(records)
