Get-MgServicePrincipalAppRoleAssignment -ServicePrincipalId $miObjectId | 
  Where-Object { $_.ResourceDisplayName -eq "Microsoft Graph" } |
  Select-Object PrincipalDisplayName, ResourceDisplayName, AppRoleId

GET https://graph.microsoft.com/v1.0/servicePrincipals?$filter=appId eq '00000003-0000-0000-c000-000000000000'&$select=appRoles

Connect-MgGraph -Scopes "Application.Read.All","AppRoleAssignment.ReadWrite.All"

# Look up the managed identity's service principal by name
$mi = Get-MgServicePrincipal -Filter "displayName eq 'id-logic-dat'"
$miObjectId = $mi.Id

# Microsoft Graph service principal
$graphSp = Get-MgServicePrincipal -Filter "appId eq '00000003-0000-0000-c000-000000000000'"

# Find the Policy.Read.All application role
$appRole = $graphSp.AppRoles | Where-Object { $_.Value -eq "Policy.Read.All" -and $_.AllowedMemberTypes -contains "Application" }

# Sanity check before assigning - all three should show a GUID
$miObjectId
$graphSp.Id
$appRole.Id

New-MgServicePrincipalAppRoleAssignment -ServicePrincipalId $miObjectId `
  -PrincipalId $miObjectId `
  -ResourceId $graphSp.Id `
  -AppRoleId $appRole.Id
