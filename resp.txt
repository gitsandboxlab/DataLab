import json
from datetime import date, datetime, timezone

import pandas as pd
from azure.identity import ClientSecretCredential
from azure.storage.blob import BlobServiceClient


def fetch_raw_json(
    account_url: str,
    container_path: str,
    tenant_id: str,
    client_id: str,
    secret: str,
    file_name: str,
    data_date: date,
) -> str:
    """
    Fetch a raw JSON blob for the given date and return it as a string.

    container_path: first segment is the container name, remainder is the subfolder
                     e.g. "raw/cap-policy" -> container "raw", subfolder "cap-policy"
    Blob path resolves to: {subfolder}/{yyyy}/{m}/{d}/{file_name}_{yyyyMMdd}.json
    """
    container_name, _, subfolder = container_path.partition("/")

    credential = ClientSecretCredential(tenant_id, client_id, secret)
    blob_service_client = BlobServiceClient(account_url=account_url, credential=credential)
    container_client = blob_service_client.get_container_client(container_name)

    blob_name = (
        f"{subfolder}/{data_date.year}/{data_date.month}/{data_date.day}/"
        f"{file_name}_{data_date:%Y%m%d}.json"
    )

    blob_client = container_client.get_blob_client(blob_name)
    return blob_client.download_blob().readall().decode("utf-8")


def parse_cap_policies(raw_json: str) -> pd.DataFrame:
    """Parse raw CA policy JSON into one row per policy."""
    policies = json.loads(raw_json)["value"]
    load_time = datetime.now(timezone.utc)

    return pd.DataFrame([{
        "PolicyId": p["id"],
        "DisplayName": p["displayName"],
        "State": p["state"],
        "LoadDateTime": load_time,
        "PolicyJson": json.dumps(p),
    } for p in policies])
