import json
import pandas as pd
from datetime import datetime, timezone

def parse_defender_recommendations(raw_json: str) -> pd.DataFrame:
    """
    Parses the raw JSON array from the fetch-defender-recommendations blob
    into a DataFrame ready for SQL insert: identifier, ingested_at, json_record.
    """
    records = json.loads(raw_json)
    ingested_at = datetime.now(timezone.utc)

    rows = []
    for record in records:
        identifier = record.get("id")
        if not identifier:
            continue
        rows.append({
            "identifier": identifier,
            "ingested_at": ingested_at,
            "json_record": json.dumps(record),
        })

    return pd.DataFrame(rows, columns=["identifier", "ingested_at", "json_record"])| Managed Identity | your UAMI |
