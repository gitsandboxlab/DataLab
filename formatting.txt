SELECT
    p.Id                                                                        AS SourceRowId,   -- remove if not applicable

    -- Top-level
    JSON_VALUE(p.PolicyJson, '$.id')                                            AS PolicyGuid,
    JSON_VALUE(p.PolicyJson, '$.displayName')                                   AS DisplayName,
    JSON_VALUE(p.PolicyJson, '$.state')                                         AS State,
    JSON_VALUE(p.PolicyJson, '$.createdDateTime')                               AS CreatedDateTime,
    JSON_VALUE(p.PolicyJson, '$.modifiedDateTime')                              AS ModifiedDateTime,

    -- conditions.users
    (SELECT STRING_AGG(value, ', ') FROM OPENJSON(p.PolicyJson, '$.conditions.users.includeUsers'))                                  AS IncludeUsers,
    (SELECT STRING_AGG(value, ', ') FROM OPENJSON(p.PolicyJson, '$.conditions.users.excludeUsers'))                                  AS ExcludeUsers,
    (SELECT STRING_AGG(value, ', ') FROM OPENJSON(p.PolicyJson, '$.conditions.users.includeGroups'))                                 AS IncludeGroups,
    (SELECT STRING_AGG(value, ', ') FROM OPENJSON(p.PolicyJson, '$.conditions.users.excludeGroups'))                                 AS ExcludeGroups,
    (SELECT STRING_AGG(value, ', ') FROM OPENJSON(p.PolicyJson, '$.conditions.users.includeRoles'))                                  AS IncludeRoles,
    (SELECT STRING_AGG(value, ', ') FROM OPENJSON(p.PolicyJson, '$.conditions.users.excludeRoles'))                                  AS ExcludeRoles,
    (SELECT STRING_AGG(value, ', ') FROM OPENJSON(p.PolicyJson, '$.conditions.users.includeGuestsOrExternalUsers.guestOrExternalUserTypes')) AS IncludeGuestTypes,
    (SELECT STRING_AGG(value, ', ') FROM OPENJSON(p.PolicyJson, '$.conditions.users.excludeGuestsOrExternalUsers.guestOrExternalUserTypes')) AS ExcludeGuestTypes,

    -- conditions.applications
    (SELECT STRING_AGG(value, ', ') FROM OPENJSON(p.PolicyJson, '$.conditions.applications.includeApplications'))                    AS IncludeApplications,
    (SELECT STRING_AGG(value, ', ') FROM OPENJSON(p.PolicyJson, '$.conditions.applications.excludeApplications'))                    AS ExcludeApplications,
    (SELECT STRING_AGG(value, ', ') FROM OPENJSON(p.PolicyJson, '$.conditions.applications.includeUserActions'))                     AS IncludeUserActions,
    (SELECT STRING_AGG(value, ', ') FROM OPENJSON(p.PolicyJson, '$.conditions.applications.includeAuthenticationContextClassReferences')) AS IncludeAuthContexts,
    JSON_VALUE(p.PolicyJson, '$.conditions.applications.applicationFilter.rule')                                                     AS ApplicationFilterRule,

    -- conditions.clientApplications (workload-identity CA only; NULL otherwise)
    (SELECT STRING_AGG(value, ', ') FROM OPENJSON(p.PolicyJson, '$.conditions.clientApplications.includeServicePrincipals'))         AS IncludeServicePrincipals,
    (SELECT STRING_AGG(value, ', ') FROM OPENJSON(p.PolicyJson, '$.conditions.clientApplications.excludeServicePrincipals'))         AS ExcludeServicePrincipals,
    JSON_VALUE(p.PolicyJson, '$.conditions.clientApplications.servicePrincipalFilter.rule')                                          AS ServicePrincipalFilterRule,

    -- conditions - flat fields
    (SELECT STRING_AGG(value, ', ') FROM OPENJSON(p.PolicyJson, '$.conditions.clientAppTypes'))                                      AS ClientAppTypes,
    (SELECT STRING_AGG(value, ', ') FROM OPENJSON(p.PolicyJson, '$.conditions.signInRiskLevels'))                                    AS SignInRiskLevels,
    (SELECT STRING_AGG(value, ', ') FROM OPENJSON(p.PolicyJson, '$.conditions.userRiskLevels'))                                      AS UserRiskLevels,
    (SELECT STRING_AGG(value, ', ') FROM OPENJSON(p.PolicyJson, '$.conditions.servicePrincipalRiskLevels'))                          AS ServicePrincipalRiskLevels,
    JSON_VALUE(p.PolicyJson, '$.conditions.insiderRiskLevels')                                                                       AS InsiderRiskLevel,  -- scalar, not an array

    -- conditions.platforms
    (SELECT STRING_AGG(value, ', ') FROM OPENJSON(p.PolicyJson, '$.conditions.platforms.includePlatforms'))                         AS IncludePlatforms,
    (SELECT STRING_AGG(value, ', ') FROM OPENJSON(p.PolicyJson, '$.conditions.platforms.excludePlatforms'))                         AS ExcludePlatforms,

    -- conditions.locations
    (SELECT STRING_AGG(value, ', ') FROM OPENJSON(p.PolicyJson, '$.conditions.locations.includeLocations'))                         AS IncludeLocations,
    (SELECT STRING_AGG(value, ', ') FROM OPENJSON(p.PolicyJson, '$.conditions.locations.excludeLocations'))                         AS ExcludeLocations,

    -- conditions.devices
    JSON_VALUE(p.PolicyJson, '$.conditions.devices.deviceFilter.mode')                                                               AS DeviceFilterMode,
    JSON_VALUE(p.PolicyJson, '$.conditions.devices.deviceFilter.rule')                                                               AS DeviceFilterRule,

    -- conditions.authenticationFlows
    JSON_VALUE(p.PolicyJson, '$.conditions.authenticationFlows.transferMethods')                                                     AS AuthFlowTransferMethods,

    -- grantControls
    JSON_VALUE(p.PolicyJson, '$.grantControls.operator')                                                                             AS GrantOperator,
    (SELECT STRING_AGG(value, ', ') FROM OPENJSON(p.PolicyJson, '$.grantControls.builtInControls'))                                  AS BuiltInControls,
    (SELECT STRING_AGG(value, ', ') FROM OPENJSON(p.PolicyJson, '$.grantControls.customAuthenticationFactors'))                      AS CustomAuthFactors,
    (SELECT STRING_AGG(value, ', ') FROM OPENJSON(p.PolicyJson, '$.grantControls.termsOfUse'))                                       AS TermsOfUse,
    JSON_VALUE(p.PolicyJson, '$.grantControls.authenticationStrength.displayName')                                                   AS AuthenticationStrengthName,

    -- sessionControls
    JSON_VALUE(p.PolicyJson, '$.sessionControls.applicationEnforcedRestrictions.isEnabled')                                          AS AppEnforcedRestrictionsEnabled,
    JSON_VALUE(p.PolicyJson, '$.sessionControls.cloudAppSecurity.isEnabled')                                                         AS CloudAppSecurityEnabled,
    JSON_VALUE(p.PolicyJson, '$.sessionControls.cloudAppSecurity.cloudAppSecurityType')                                              AS CloudAppSecurityType,
    JSON_VALUE(p.PolicyJson, '$.sessionControls.signInFrequency.isEnabled')                                                          AS SignInFrequencyEnabled,
    JSON_VALUE(p.PolicyJson, '$.sessionControls.signInFrequency.value')                                                              AS SignInFrequencyValue,
    JSON_VALUE(p.PolicyJson, '$.sessionControls.signInFrequency.type')                                                               AS SignInFrequencyType,
    JSON_VALUE(p.PolicyJson, '$.sessionControls.signInFrequency.authenticationType')                                                 AS SignInFrequencyAuthType,
    JSON_VALUE(p.PolicyJson, '$.sessionControls.signInFrequency.frequencyInterval')                                                  AS SignInFrequencyInterval,
    JSON_VALUE(p.PolicyJson, '$.sessionControls.persistentBrowser.isEnabled')                                                        AS PersistentBrowserEnabled,
    JSON_VALUE(p.PolicyJson, '$.sessionControls.persistentBrowser.mode')                                                             AS PersistentBrowserMode,
    JSON_VALUE(p.PolicyJson, '$.sessionControls.continuousAccessEvaluation.mode')                                                    AS ContinuousAccessEvalMode,
    JSON_VALUE(p.PolicyJson, '$.sessionControls.disableResilienceDefaults')                                                          AS DisableResilienceDefaults,
    JSON_VALUE(p.PolicyJson, '$.sessionControls.secureSignInSession.isEnabled')                                                      AS SecureSignInSessionEnabled

FROM dbo.ConditionalAccessPolicies AS p;
